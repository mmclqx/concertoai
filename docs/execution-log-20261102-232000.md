## 九、附录：探针源码（明天修完照抄重跑，做 before/after 对比）

```ts
/** K18 探针（临时，用完即删）：同一固定时刻、两种 locale 下的档位标注与整行输出。 */
import { priceTierSummary } from './src/cli/balance';
import { configureHolidayCalendar, resetHolidayCalendar } from './src/core/holidays';
import { setLocale } from './src/i18n';
import { tierLabel } from './src/i18n'; // ← L2 迁移后从这里取（修前取自 './src/model/models'）

// 周三 20:00 = 空闲档 / 周三 10:00 = 高峰档（固定时刻，不依赖「今天」）
const OFFPEAK_AT = new Date(2026, 9, 28, 20, 0, 0);
const PEAK_AT = new Date(2026, 9, 28, 10, 0, 0);

configureHolidayCalendar({ holidays: [] }); // 有日历 → 不附加「未配置节假日表」提示

for (const locale of ['zh', 'en'] as const) {
  setLocale(locale);
  console.log(`=== locale=${locale} ===`);
  console.log('tierLabel(peak)    =', tierLabel('peak'));
  console.log('tierLabel(offpeak) =', tierLabel('offpeak'));
  console.log('--- priceTierSummary(offpeak) ---');
  console.log(priceTierSummary(OFFPEAK_AT));
  console.log('--- priceTierSummary(peak) ---');
  console.log(priceTierSummary(PEAK_AT));
  console.log('--- 含 CJK 字符? ---', /[\u4e00-\u9fff]/.test(priceTierSummary(OFFPEAK_AT)));
}

resetHolidayCalendar();
setLocale('zh');
```

**修后期望**：`locale=en` 两行 `tierLabel` 为 `peak hours` / `off-peak hours`，两段 `priceTierSummary` 的 `含 CJK` 均为 `false`；`locale=zh` 保持原样（除补空格）。
